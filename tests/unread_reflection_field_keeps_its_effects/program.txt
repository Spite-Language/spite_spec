var console = Console()

func Written() {
    var monster = Monster()
    console.print(monster.health)
    var attribute = monster.attributes["health"]
    attribute.remember_through(loud, 4)
    attribute.remember_through(making, 5)
    var names = List<String>()
    attribute.remember_count(names)
    var appended = names.count()
    console.print(appended)
    attribute.remember_a_shout()
    attribute.remember_quietly(6)
    console.print("done")
}

func loud(value: Integer): Integer {
    console.print("written")
    return value
}

func making(value: Integer): Integer {
    console.print("made")
    var made = Monster()
    return made.health + value
}
